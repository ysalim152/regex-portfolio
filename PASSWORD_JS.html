Validation de Force de Mot de Passe (JavaScript)

Retour à l'accueil

Le code JavaScript utilisé

Ce script évalue la force d'un mot de passe en temps réel en se basant sur plusieurs critères validés par des expressions régulières :

  • Au moins 8 caractères
  • Contient une minuscule (/[a-z]/)
  • Contient une majuscule (/[A-Z]/)
  • Contient un chiffre (/[0-9]/)
  • Contient un caractère spécial (/[^a-zA-Z0-9]/)

const passwordInput = document.getElementById('password');
const strengthMessage = document.getElementById('password-strength');
const strengthBarFill = document.getElementById('strength-bar-fill');

passwordInput.addEventListener('input', () => {
    const password = passwordInput.value;
    let score = 0;

    if (password.length >= 8) score++;
    if (/[a-z]/.test(password)) score++;
    if (/[A-Z]/.test(password)) score++;
    if (/[0-9]/.test(password)) score++;
    if (/[^a-zA-Z0-9]/.test(password)) score++;

    let strength = { text: '', color: 'transparent', width: '0%' };

    switch (score) {
        case 0: case 1: case 2:
            strength = { text: 'Faible', color: 'var(--invalid-color, #d9534f)', width: '25%' };
            break;
        case 3:
            strength = { text: 'Moyen', color: 'orange', width: '50%' };
            break;
        case 4:
            strength = { text: 'Bon', color: '#5c940d', width: '75%' };
            break;
        case 5:
            strength = { text: 'Excellent', color: 'var(--valid-color, #5cb85c)', width: '100%' };
            break;
    }
    
    if (password.length === 0) {
        strength = { text: '', color: 'transparent', width: '0%' };
    }

    strengthMessage.textContent = strength.text;
    strengthBarFill.style.backgroundColor = strength.color;
    strengthBarFill.style.width = strength.width;
});
